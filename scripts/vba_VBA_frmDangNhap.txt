' ===========================================================================
' Form: frmDangNhap (v6.2 - Smart Invoice API Login)
'
' Updated to use the local Webapp XML Smart Invoice API for authentication.
' The webapp handles CAPTCHA solving automatically via:
'   - Vector path signature matching (instant, 100% accurate for known chars)
'   - ddddocr OCR fallback for unknown character variants
'
' Login modes:
'   1) SMART (default): Via webapp /API/login endpoint (auto-captcha)
'   2) DIRECT (fallback): Via GDT portal directly (local SVG detection)
' ===========================================================================

Dim url As String, res As String, blnCaptcha As Boolean
Dim arrUsers
Dim blnSmartMode As Boolean  ' True = use webapp proxy, False = direct GDT

    Me.cmdHienMK.Visible = True

    Me.cmdHienMK.Visible = False
    Me.cmdDauMK.Visible = True
End Sub

Private Sub lsbUsers_Change()
    Dim user As String, pass As String, tenDV As String
    user = Me.lsbUsers.list(Me.lsbUsers.ListIndex)
    pass = Me.lsbUsers.list(Me.lsbUsers.ListIndex, 1)
    tenDV = Me.lsbUsers.list(Me.lsbUsers.ListIndex, 2)
    Me.txtUser = user: Me.txtPass = pass: Me.txtTenDV = tenDV: Me.lblTenDV.Caption = tenDV
End Sub

Private Sub lsbUsers_Exit(ByVal Cancel As MSForms.ReturnBoolean)
    Me.lsbUsers.Visible = False
End Sub

Private Sub txtPass_Exit(ByVal Cancel As MSForms.ReturnBoolean)
    Me.lsbUsers.Visible = False
    
    ' In Smart mode, auto-captcha is handled by the webapp - no manual captcha needed
    If blnSmartMode Then
        blnCaptcha = True
        ' Hide captcha controls since webapp handles it automatically
        On Error Resume Next
        Me.txtNhapCaptcha.Visible = False
        Me.lblCaptcha.Visible = False
        On Error GoTo 0
    Else
        ' Legacy mode: fetch and display captcha for manual/auto detection
        If layCaptcha Then
            blnCaptcha = True
        Else
            blnCaptcha = False
        End If
    End If
    
    Me.cmdDangNhap.SetFocus
End Sub

Private Sub txtUser_AfterUpdate()
    On Error Resume Next    'Vuot loi could not set property visible
    Me.lsbUsers.Visible = False
End Sub

Private Sub txtUser_Change()
    Me.lsbUsers.Visible = True
End Sub

Private Sub UserForm_Initialize()
    Me.lblThanhCong.Visible = False
    Me.lblThatBai.Visible = False
    Me.cmdHienMK.Visible = True
    Me.cmdDauMK.Visible = False
    layUserPass
    createXmlHttpObj
    
    ' Auto-detect if webapp is running to choose login mode
    blnSmartMode = TestWebappConnection()
    
    If blnSmartMode Then
        ' Smart mode: webapp handles captcha automatically
        Me.Caption = "Dang nhap - Auto Captcha (Webapp)"
        blnCaptcha = True   ' Always ready since captcha is auto-solved
        
        ' Hide captcha input controls
        On Error Resume Next
        Me.txtNhapCaptcha.Visible = False
        Me.lblCaptcha.Visible = False
        On Error GoTo 0
        
        Debug.Print "[frmDangNhap] Smart mode: Webapp connected at " & GetWebappBaseUrl()
    Else
        ' Legacy mode: manual/SVG captcha detection
        Me.Caption = "Dang nhap - Captcha thu cong"
        Debug.Print "[frmDangNhap] Legacy mode: Webapp not available, using direct GDT login"
    End If
End Sub

Private Sub txtPass_AfterUpdate()
    If Not blnSmartMode Then
        ' Only fetch captcha in legacy mode
        If layCaptcha Then
            blnCaptcha = True
        Else
            blnCaptcha = False
        End If
    End If
End Sub

Private Sub cmdDangNhap_Click()
    
    If Me.txtUser = "" Or Me.txtPass = "" Then
        MsgBoxUni "Vui l" & ChrW(242) & "ng nh" & ChrW(7853) & "p t" & ChrW(234) & "n " & ChrW(273) & ChrW(259) & "ng nh" & ChrW(7853) & "p v" & ChrW(224) & " m" & ChrW(7853) & "t kh" & ChrW(7849) & "u.", vbCritical
        Exit Sub
    End If
    
    ' ── SMART MODE: Login via Webapp Smart Invoice API ──
    If blnSmartMode Then
        Call SmartModeLogin
        Exit Sub
    End If
    
    ' ── LEGACY MODE: Direct GDT login with local captcha ──
    If blnCaptcha = False Then Exit Sub
    Call LegacyModeLogin
    
End Sub


Private Sub SmartModeLogin()
    ' Login via the local Webapp XML Smart Invoice API.
    ' The webapp handles captcha fetching, solving, and GDT authentication.
    
    Me.lblThanhCong.Visible = False
    Me.lblThatBai.Visible = False
    
    Dim token As String
    
    ' Disable button during login to prevent double-clicks
    Me.cmdDangNhap.Enabled = False
    DoEvents
    
    ' Call SmartLogin from modSmartInvoiceLogin
    token = SmartLogin(Me.txtUser, Me.txtPass)
    
    Me.cmdDangNhap.Enabled = True
    
    If Len(token) > 0 Then
        ' SUCCESS
        cToken = token
        bearer = token
        sMST = Me.txtUser
        Me.txtTenDV = layTenDV
        
        Me.lblThanhCong.Visible = True
        Me.lblThatBai.Visible = False
        Call saveUserPass
        Application.OnTime Time + TimeValue("00:00:01"), "closeForm"
    Else
        ' FAILED - offer fallback to direct mode
        Me.lblThanhCong.Visible = False
        Me.lblThatBai.Visible = True
        
        Dim sMsg As String, ret As Long
        sMsg = ChrW(272) & ChrW(259) & "ng nh" & ChrW(7853) & "p qua Webapp kh" & ChrW(244) & "ng th" & ChrW(224) & "nh c" & ChrW(244) & "ng." & vbCrLf
        sMsg = sMsg & "B" & ChrW(7841) & "n c" & ChrW(243) & " mu" & ChrW(7889) & "n th" & ChrW(7917) & " " & ChrW(273) & ChrW(259) & "ng nh" & ChrW(7853) & "p tr" & ChrW(7921) & "c ti" & ChrW(7871) & "p v" & ChrW(224) & "o trang Thu" & ChrW(7871) & "?"
        ret = MsgBox(sMsg, vbYesNo + vbQuestion, "Smart Login")
        
        If ret = vbYes Then
            ' Switch to legacy mode
            blnSmartMode = False
            Me.Caption = "Dang nhap - Captcha thu cong"
            
            ' Show captcha controls
            On Error Resume Next
            Me.txtNhapCaptcha.Visible = True
            Me.lblCaptcha.Visible = True
            On Error GoTo 0
            
            ' Fetch captcha
            If layCaptcha Then
                blnCaptcha = True
            Else
                blnCaptcha = False
            End If
        End If
    End If
End Sub


Private Sub LegacyModeLogin()
    ' Original direct GDT login with local SVG captcha detection.
    
    url = "https://hoadondientu.gdt.gov.vn:30000/security-taxpayer/authenticate"
    Call httpOption(url)
    
    Dim formData As String
    formData = "{""username"":""" & Me.txtUser & """,""password"":""" & Me.txtPass & """,""cvalue"":""" & Me.txtNhapCaptcha & """,""ckey"":""" & cKey & """}"
    createXmlHttpObj
    res = httpPost(url, formData)
    
    Set js = CreateObject("Scripting.Dictionary")
    Set js = JsonConverter.ParseJSON(res)
    
    If InStr(1, res, "message") > 0 Then
        If InStr(1, res, "captcha") Then
            MsgBoxUni js("message") & vbCrLf & "Vui l" & ChrW(242) & "ng nh" & ChrW(7853) & "p l" & ChrW(7841) & "i.", vbCritical, "Thông báo"
        Else
            MsgBoxUni js("message")
        End If
        
        Me.txtNhapCaptcha = ""
        Call layCaptcha
        
        Me.lblThanhCong.Visible = False
        Me.lblThatBai.Visible = True
        Exit Sub
    Else
        cToken = js("token")
        bearer = cToken
        sMST = Me.txtUser
        Me.txtTenDV = layTenDV
        
        Me.lblThanhCong.Visible = True
        Me.lblThatBai.Visible = False
        Call saveUserPass
        Application.OnTime Time + TimeValue("00:00:01"), "closeForm"
    End If
    
End Sub

Function layCaptcha() As Boolean
    layCaptcha = True
    'Kiem tra ket noi internet
    If GetInternetConnectedState = False Then
        MsgBoxUni "Kh" & ChrW(244) & "ng c" & ChrW(243) & " k" & ChrW(7871) & "t n" & ChrW(7889) & "i internet.", vbCritical, "No internet"
        layCaptcha = False
        Exit Function
    End If
    
    Call createXmlHttpObj
    Dim cContent As String, sMsg As String
    url = "https://hoadondientu.gdt.gov.vn:30000/captcha"
    If httpGet(url, "", res) Then
        Set js = JsonConverter.ParseJSON(res)
        cKey = js("key")
        cContent = js("content")
        
        'Luu noi dung vao file .txt
        SaveStringToTextFile cContent
        
        DoEvents
        Me.txtNhapCaptcha = detectSVGCaptcha(res)
        
    Else
        sMsg = "C" & ChrW(243) & " l" & ChrW(7895) & "i k" & ChrW(7871) & "t n" & ChrW(7889) & "i v" & ChrW(224) & " l" & ChrW(7845) & "y d" & ChrW(7919) & " li" & ChrW(7879) & "u."
        sMsg = sMsg & vbCrLf & "Vui l" & ChrW(242) & "ng kh" & ChrW(7903) & "i " & ChrW(273) & ChrW(7897) & "ng l" & ChrW(7841) & "i " & ChrW(7913) & "ng d" & ChrW(7909) & "ng."
        MsgBoxUni sMsg, vbCritical
        layCaptcha = False
    End If
    
End Function

Sub SaveStringToTextFile(textData As String)
    Dim filePath As String
    Dim fileNum As Integer
    
    filePath = Environ$("USERPROFILE") & "\Documents\" & "myCaptcha.txt"
    If Dir(Environ$("USERPROFILE") & "\Documents\" & "myCaptcha.svg") <> "" Then
        On Error Resume Next    'permission denied
        Kill Environ$("USERPROFILE") & "\Documents\" & "myCaptcha.svg"
    End If
    
    fileNum = FreeFile
    Open filePath For Output As fileNum
    Print #fileNum, textData
    Close fileNum
    Name filePath As Replace(filePath, ".txt", ".svg")
End Sub

Sub layUserPass()
    arrUsers = Sheets("LinkTraCuu").Range("Q2:S" & Sheets("LinkTraCuu").Cells(Rows.Count, "Q").End(xlUp).row).Value
    Me.lsbUsers.list = arrUsers
    Me.lsbUsers.Width = Me.txtUser.Width
    Me.lsbUsers.Height = 100
    Me.lsbUsers.Left = Me.txtUser.Left
    Me.lsbUsers.Top = Me.txtUser.Top + Me.txtUser.Height
End Sub

Sub saveUserPass()
    Dim X As Long, lr As Long
    For X = 1 To UBound(arrUsers, 1)
        If Me.txtUser = arrUsers(X, 1) Then
            Exit Sub
        End If
    Next
    'Luu user/pass moi
    With Sheets("LinkTraCuu")
        lr = .Cells(Rows.Count, "Q").End(xlUp).row
        .Range("Q" & lr + 1).Value = Me.txtUser
        .Range("R" & lr + 1).Value = Me.txtPass
        .Range("S" & lr + 1).Value = Me.txtTenDV
        ThisWorkbook.Save
    End With

End Sub

Function layTenDV() As String
    url = "https://hoadondientu.gdt.gov.vn:30000/security-taxpayer/profile"
    If httpGet(url, bearer, res) Then
        Set js = JsonConverter.ParseJSON(res)
        layTenDV = js("name")
    Else
        layTenDV = ""
    End If
End Function
