Function correctDate(ByRef textDate As String, ByRef err As Boolean) As Date

    Dim ngay As Long, thang As Long, nam As Long

    Dim dtNgayNhap As Date, dtNgayCuoiThang As Date

    

    If Len(textDate) < 10 Or Len(textDate) > 10 Then

        correctDate = Date

        err = True

        Exit Function

    ElseIf Mid(textDate, 3, 1) <> "/" Or Mid(textDate, 6, 1) <> "/" Then

        correctDate = Date

        err = True

        Exit Function

    End If

    

    ngay = CLng(Left(textDate, 2))

    thang = CLng(Mid(textDate, 4, 2))

    nam = CLng(Right(textDate, 4))

    

    If thang > 12 Then thang = 12

    

    dtNgayNhap = DateSerial(nam, thang, ngay)

    dtNgayCuoiThang = DateSerial(nam, thang + 1, 0)

    If dtNgayNhap > dtNgayCuoiThang Then dtNgayNhap = dtNgayCuoiThang

    

    correctDate = CStr(dtNgayNhap)

    

End Function

Sub taiHoaDon_Total(Optional rowTotalstart As Long = 3, Optional rowDetailStart As Long = 3)

    Dim item As Object

    Dim arrHDChiTiet_tmp(1000, 5)

    

    'Tao bang tra cac dieu kien tim kiem

    Call LinkTraCuu

    Call tenCotTraCuu

    arrTrangThai = Sheets("LinkTraCuu").Range("I2:I8").Value

    arrKQKTHoaDon = Sheets("LinkTraCuu").Range("O2:O11").Value

    stt = 1

    row = rowTotalstart: row_ct = rowDetailStart

    

    'Tham so cho request

    size = 50

    tthai = Me.cboTTHD

        url_sco = "https://hoadondientu.gdt.gov.vn:30000/sco-query/invoices/sold?sort="

        loaiHD = 2 'ban

    n = 0   'n: so luong HD

    'Duyet tung khoan thoi gian de trich xuat hoa don

    For k = 1 To UBound(arrDate)

        If tthai = "All" Then 'Tat ca

            'Hd mua va ban giong nhau

            If ttxly = "All" Then  'Tat ca

                search = "tdlap=ge=" & Format(arrDate(k, 1), "dd/mm/yyyy") & "T00:00:00;tdlap=le=" & Format(arrDate(k, 2), "dd/mm/yyyy") & "T23:59:59"

            Else

                search = "tdlap=ge=" & Format(arrDate(k, 1), "dd/mm/yyyy") & "T00:00:00;tdlap=le=" & Format(arrDate(k, 2), "dd/mm/yyyy") & "T23:59:59;ttxly==" & ttxly

            End If

        Else

            If ttxly = "All" Then

                search = "tdlap=ge=" & Format(arrDate(k, 1), "dd/mm/yyyy") & "T00:00:00;tdlap=le=" & Format(arrDate(k, 2), "dd/mm/yyyy") & "T23:59:59;tthai==" & tthai

            End If

        End If

        url2 = url & sort & "&size=" & size & "&search=" & search

nextPage:
        ' === Resilient Search Retry Loop ===
        Dim searchAttempt As Long
        Dim searchSuccess As Boolean
        searchSuccess = False
        
        For searchAttempt = 1 To 3
            If httpGet(url2, bearer, res, SilentMode:=True) And Len(res) > 0 Then
                If InStr(1, res, "error") = 0 Then
                    searchSuccess = True
                    Exit For
                End If
            End If
            
            Debug.Print "[taiHoaDon_Total] Search attempt " & searchAttempt & " failed for URL: " & url2
            Call WriteAuditLog("WARNING", "SEARCH_RETRY", "Attempt " & searchAttempt & " failed for period: " & arrDate(k, 1) & " - " & arrDate(k, 2))
            
            If searchAttempt < 3 Then
                Sleep (1000 * searchAttempt)
            End If
        Next searchAttempt
        
        If Not searchSuccess Then
            errMsg = errMsg & "Loi tai hoa don tong hop ngay: " & arrDate(k, 1) & " - " & arrDate(k, 2) & vbCrLf
            Call WriteAuditLog("ERROR", "SEARCH_FAIL", "Failed to search invoices for period: " & arrDate(k, 1) & " - " & arrDate(k, 2))
            If getStatus = 429 Or getStatus = 500 Then GoTo errLog
            GoTo nextDatePeriod
        End If

        

        Set js = JsonConverter.ParseJSON(res)

        If InStr(1, res, "error") > 0 Then

            arrHDChiTiet_tmp(n, 5) = ISODATE(item("tdlap"))

            n = n + 1

        Next

        

        'Tai nhieu trang

            url2 = url & sort & "&size=" & size & "&state=" & js("state") & "&search=" & search

            GoTo nextPage 'Quay lai lay du lieu tiep trang 2 (>50)...

        End If

        

           '******************************************

        '// Chay sco query de lay hd tu may tinh tien

        url2 = url_sco & sort & "&size=" & size & "&search=" & search

nextPage_sco:
        ' === Resilient SCO Search Retry Loop ===
        Dim scoSearchAttempt As Long
        Dim scoSearchSuccess As Boolean
        scoSearchSuccess = False
        
        For scoSearchAttempt = 1 To 3
            If httpGet(url2, bearer, res, SilentMode:=True) And Len(res) > 0 Then
                If InStr(1, res, "error") = 0 Then
                    scoSearchSuccess = True
                    Exit For
                End If
            End If
            
            Debug.Print "[taiHoaDon_Total] SCO Search attempt " & scoSearchAttempt & " failed for URL: " & url2
            Call WriteAuditLog("WARNING", "SCO_SEARCH_RETRY", "Attempt " & scoSearchAttempt & " failed for period: " & arrDate(k, 1) & " - " & arrDate(k, 2))
            
            If scoSearchAttempt < 3 Then
                Sleep (1000 * scoSearchAttempt)
            End If
        Next scoSearchAttempt
        
        If Not scoSearchSuccess Then
            errMsg = errMsg & "Loi tai hoa don tong hop ngay: " & arrDate(k, 1) & " - " & arrDate(k, 2) & vbCrLf
            Call WriteAuditLog("ERROR", "SCO_SEARCH_FAIL", "Failed to search SCO invoices for period: " & arrDate(k, 1) & " - " & arrDate(k, 2))
            If getStatus = 429 Or getStatus = 500 Then GoTo errLog
            GoTo nextDatePeriod
        End If

        

        Set js = JsonConverter.ParseJSON(res)

        If InStr(1, res, "error") > 0 Then

            'MsgBoxUni js("error") & vbCrLf & vbCrLf & "Vui long khoi dong lai ung dung"

            errMsg = errMsg & "Loi ket noi: " & js("error") & vbCrLf

            GoTo nextDatePeriod

        End If

        

        ghiExcel_TongHop res, row, loaiHD, stt

        

        'Lay tham so HD chi tiet

        For Each item In js("datas")

            arrHDChiTiet_tmp(n, 0) = item("nbmst") & ""

            arrHDChiTiet_tmp(n, 1) = item("khhdon")

            arrHDChiTiet_tmp(n, 2) = item("shdon")

            arrHDChiTiet_tmp(n, 3) = item("khmshdon")

            arrHDChiTiet_tmp(n, 4) = 2  'sco-query

            arrHDChiTiet_tmp(n, 5) = ISODATE(item("tdlap"))

            n = n + 1

        'Tai nhieu trang

errLog:

    End If

    

    MsgBox "Xong."



cleanup:

    Erase arrHDChiTiet

    'Unload Me

End Sub



Sub lietKeThoiGian()

    Dim startYear As Long, startMonth As Long, startDay As Long, endYear As Long, endMonth As Long, endDay As Long

    Dim dtNgayCuoiThang As Date, dtNgayDauThang As Date, dtDenNgay As Date

    Dim tempArr()   'De redim arrDate

    

    startYear = Right(Me.txtTuNgay, 4)

    startMonth = Mid(Me.txtTuNgay, 4, 2)

    startDay = Left(Me.txtTuNgay, 2)

    

    endYear = Right(Me.txtDenNgay, 4)

    endMonth = Mid(Me.txtDenNgay, 4, 2)

    endDay = Left(Me.txtDenNgay, 2)

    

    dtNgayDauThang = DateSerial(startYear, startMonth, startDay)

    dtNgayCuoiThang = DateSerial(startYear, startMonth + 1, 0)

    dtDenNgay = DateSerial(endYear, endMonth, endDay)

    i = 1

    ReDim arrDate(60, 2)    '5 nam

    Do While dtNgayCuoiThang <= dtDenNgay

        

        arrDate(i, 1) = dtNgayDauThang: arrDate(i, 2) = dtNgayCuoiThang

        startMonth = startMonth + 1

        'Qua nam ke tiep

        If startMonth > 12 Then

            startMonth = 1

            startYear = startYear + 1

        End If

        dtNgayDauThang = DateSerial(startYear, startMonth, 1)

        dtNgayCuoiThang = DateSerial(startYear, startMonth + 1, 0)

        i = i + 1

    Loop

    If dtNgayDauThang <= dtDenNgay Then

        arrDate(i, 1) = dtNgayDauThang: arrDate(i, 2) = dtDenNgay

    Else

        i = i - 1

    End If

    

    ReDim tempArr(1 To i, 1 To 2)

    Dim k As Long

    For k = 1 To i

        tempArr(k, 1) = arrDate(k, 1)

        tempArr(k, 2) = arrDate(k, 2)

    Next

    

    arrDate = tempArr

    

    'For k = 1 To UBound(arrDate)

        'Debug.Print arrDate(k, 1), arrDate(k, 2)

    'Next

End Sub

Sub taiXML_zip(soHD As Long)

    Dim nbmst As String, khhdon As String, shdon As String, khmshdon As String

    Call WriteAuditLog("INFO", "TAI_XML_START", "Created save folder: " & oSaveUnzipFolder & " for " & soHD & " invoices.")
    
    '-----------------------------------/
    Application.Cursor = xlWait
    DoEvents

    Dim client As Object, stream As Object
    Set client = GetxmlHttp() ' Reuse global HTTP Keep-Alive client
    Set stream = CreateObject("ADODB.Stream")

    For m = 0 To soHD - 1
        If arrHDChiTiet(m, 4) = 1 Then  'query
            urlXml = "https://hoadondientu.gdt.gov.vn:30000/query/invoices/export-xml?"
        Else    'sco-query
            urlXml = "https://hoadondientu.gdt.gov.vn:30000/sco-query/invoices/export-xml?"
        End If

        nbmst = arrHDChiTiet(m, 0)
        khhdon = arrHDChiTiet(m, 1)
        shdon = arrHDChiTiet(m, 2)
        khmshdon = arrHDChiTiet(m, 3)
        payload = "nbmst=" & nbmst & "&khhdon=" & khhdon & "&shdon=" & shdon & "&khmshdon=" & khmshdon
        urlXml = urlXml & payload

        ' Update status label for user feedback
        Me.lblStatus.Caption = "Dang tai XML: " & khhdon & "_" & shdon & " (" & (m + 1) & "/" & soHD & ")"
        DoEvents

        ' === Resilient Retry Loop with Exponential Backoff ===
        Dim attempt As Long
        Dim success As Boolean
        Dim httpStatus As Long
        success = False

        For attempt = 1 To 3
            On Error Resume Next
            client.Open "GET", urlXml, False
            client.setRequestHeader "User-Agent", "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/128.0.0.0 Safari/537.36"
            client.setRequestHeader "Accept", "application/zip,application/octet-stream,*/*"
            client.setRequestHeader "Accept-Encoding", "gzip"
            client.setRequestHeader "Authorization", "Bearer " & bearer
            client.send
            
            httpStatus = client.Status
            Dim errNum As Long
            errNum = err.Number
            On Error GoTo EH

            If errNum = 0 And httpStatus = 200 Then
                ' Double-check Content-Type to make sure GDT didn't return an error page/JSON as 200 OK
                Dim contentType As String
                On Error Resume Next
                contentType = LCase(client.getResponseHeader("Content-Type"))
                On Error GoTo EH
                
                If InStr(contentType, "zip") > 0 Or InStr(contentType, "octet-stream") > 0 Or Len(contentType) = 0 Then
                    success = True
                    Exit For
                Else
                    Debug.Print "[taiXML_zip] Got HTML/JSON instead of ZIP: " & contentType
                End If
            End If

            Debug.Print "[taiXML_zip] Attempt " & attempt & " failed for " & khhdon & "_" & shdon & " (Status: " & httpStatus & ")"
            If attempt < 3 Then
                Sleep (1000 * attempt)
            End If
        Next attempt

        If success Then
            stream.Type = 1 ' Binary
            stream.Open
            stream.Write client.responseBody ' Write the binary response to the stream
            filePath = Me.txtXMLFolderPath & khhdon & "_" & shdon & ".zip"
            stream.SaveToFile filePath, 2 ' Overwrite
            stream.Close
            DoEvents

            '//Unzip file vua tai, chi lay file xml/html
            Call Unzip(filePath, oSaveUnzipFolder)

            '// Upload directly to local Webapp XML database if connection is active
            If TestWebappConnection() Then
                Dim zipFileName As String
                zipFileName = khhdon & "_" & shdon & ".zip"
                Call UploadXMLToWebappBytes(client.responseBody, sMST, zipFileName, bearer)
            End If
        Else
            ' Log failure and proceed to next invoice (do not block user with MsgBox)
            Dim failDetail As String
            If httpStatus = 500 Then
                failDetail = "500 Internal Error (ho so goc khong ton tai)"
                errMsg = errMsg & "- Khong ton tai ho so goc cua hoa don: " & khhdon & "_" & shdon & vbCrLf
            ElseIf httpStatus = 429 Then
                failDetail = "429 Too Many Requests (bi chan boi GDT)"
                errMsg = errMsg & "- Qua nhieu yeu cau: " & khhdon & "_" & shdon & vbCrLf
            Else
                failDetail = "Status " & httpStatus & " / Loi ket noi"
                errMsg = errMsg & "- Loi tai file hoac loi ket noi: " & khhdon & "_" & shdon & vbCrLf
            End If
            
            Call WriteAuditLog("ERROR", "TAI_XML_FAIL", "Failed to download " & khhdon & "_" & shdon & " after 3 attempts: " & failDetail)
            GoTo nextInvoice
        End If

nextInvoice:
        Sleep Me.txtSleep
    Next m

EH_Exit:
    If errMsg <> "" Then
        ghiLog errMsg
    End If
    
    Set stream = Nothing
    Application.Cursor = xlDefault
    Exit Sub

EH:
    Application.Cursor = xlDefault
    Dim errDesc As String
    errDesc = "Ma loi: " & err.Number & vbCrLf & "Noi dung: " & err.Description
    Call WriteAuditLog("ERROR", "TAI_XML_EXCEPTION", "Exception in taiXML_zip: " & errDesc)
    Resume EH_Exit
End Sub
