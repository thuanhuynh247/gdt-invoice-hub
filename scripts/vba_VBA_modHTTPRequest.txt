Public xmlHttp As Object
Public cookieHeader As String
Public errMsg As String

' ===========================================================================
' Reusable ServerXMLHTTP initialization to enable HTTP Keep-Alive
' ===========================================================================
Public Function GetxmlHttp() As Object
    If xmlHttp Is Nothing Then
        Set xmlHttp = CreateObject("MSXML2.ServerXMLHTTP.6.0")
        ' Set timeouts: Resolve, Connect, Send, Receive (in milliseconds)
        ' Resolve = 15s, Connect = 15s, Send = 30s, Receive = 60s
        xmlHttp.setTimeouts 15000, 15000, 30000, 60000
    End If
    Set GetxmlHttp = xmlHttp
End Function

Public Function createXmlHttpObj()
    Set xmlHttp = GetxmlHttp()
End Function

' ===========================================================================
' HTTP POST with optional SilentMode to suppress batch execution dialogs
' ===========================================================================
Function httpPost(ByVal url As String, ByVal formData As String, Optional ByVal SilentMode As Boolean = False) As String
    On Error GoTo EH

    Dim client As Object
    Set client = GetxmlHttp()
    
    With client
        .Open "POST", url, False
        .setRequestHeader "User-Agent", "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/128.0.0.0 Safari/537.36"
        .setRequestHeader "Content-type", "application/json; charset=UTF-8"
        .send (formData)
        
        httpPost = .responseText
        
        ' Save cookies if returned
        On Error Resume Next
        cookieHeader = .getResponseHeader("Set-Cookie")
        On Error GoTo EH
    End With
    
EH_Exit:
    Exit Function
    
EH:
    Dim errDesc As String
    errDesc = "POST Error: " & err.Number & " - " & err.Description
    Debug.Print "[httpPost] " & errDesc
    
    If Not SilentMode Then
        Select Case err.Number
            Case -2147012889
                MsgBox "Loi ket noi Server hoac sai dia chi Web", vbExclamation, "Thông báo - Http Post error"
            Case -2147012894
                MsgBoxUni "Qu" & ChrW(225) & " th" & ChrW(7901) & "i gian k" & ChrW(7871) & "t n" & ChrW(7889) & "i v" & ChrW(7899) & "i m" & ChrW(225) & "y ch" & ChrW(7911) & " Web." & vbCrLf _
                & "Ki" & ChrW(7875) & "m tra " & ChrW(273) & ChrW(432) & ChrW(7901) & "ng truy" & ChrW(7873) & "n Internet v" & ChrW(224) & " th" & ChrW(7917) & " k" & ChrW(7871) & "t n" & ChrW(7889) & "i l" & ChrW(7841) & "i.", vbExclamation, "Thông báo - Http Post error"
            Case Else
                MsgBox "ErrNum: " & err.Number & vbCrLf & "Err description: " & err.Description, vbExclamation, "Thông báo"
        End Select
    End If
    
    ' Record to Audit Trail silently if taxpayer session is active
    If Len(sMST) > 0 Then
        Call WriteAuditLog("ERROR", "HTTP_POST", errDesc & " (URL: " & url & ")")
    End If
    
    httpPost = "Error"
    Resume EH_Exit
End Function

' ===========================================================================
' HTTP GET with optional SilentMode to suppress batch execution dialogs
' ===========================================================================
Function httpGet(ByVal url As String, bearer As String, res As String, Optional ByVal SilentMode As Boolean = False) As Boolean
    On Error GoTo EH
    
    Dim client As Object
    Set client = GetxmlHttp()
    
    With client
        .Open "GET", url, False
        .setRequestHeader "User-Agent", "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/128.0.0.0 Safari/537.36"
        .setRequestHeader "Accept", "text/html,application/xhtml+xml,application/xml;q=0.9,image/webp,image/apng,*/*;q=0.8,application/signed-exchange;v=b3;q=0.9"
        .setRequestHeader "Accept-Encoding", "gzip;q=1.0"
        .setRequestHeader "Content-type", "application/json; charset=UTF-8"
        
        If Len(bearer) > 0 Then
            .setRequestHeader "Authorization", "Bearer " & bearer
        End If
        
        .send
        res = .responseText
        
        If err.Number = 0 Then
            getStatus = .Status
            If getStatus <> "300" And getStatus <> "200" Then
                Dim statusMsg As String
                statusMsg = "HTTP Status: " & getStatus
                Debug.Print "[httpGet] GDT Server returned status: " & getStatus
                
                If Not SilentMode Then
                    Select Case getStatus
                    Case 429
                        MsgBoxUni "L" & ChrW(7895) & "i 429 - G" & ChrW(7917) & "i qu" & ChrW(225) & " nhi" & ChrW(7873) & "u y" & ChrW(234) & "u c" & ChrW(7847) & "u t" & ChrW(7899) & "i m" & ChrW(225) & "y ch" & ChrW(7911) & " Web.", vbExclamation, "Http Get Error 429"
                    Case 500
                        MsgBoxUni "L" & ChrW(7895) & "i web server t" & ChrW(7915) & " ch" & ChrW(7889) & "i k" & ChrW(7871) & "t n" & ChrW(7889) & "i.", vbExclamation, "Http request error 500"
                    Case Else
                        ' Silent debug
                    End Select
                End If
                
                If Len(sMST) > 0 Then
                    Call WriteAuditLog("ERROR", "HTTP_GET", statusMsg & " for URL: " & url)
                End If
                
                httpGet = False
            Else
                httpGet = True
            End If
        End If
    End With
    
EH_Exit:
    Exit Function
    
EH:
    Dim errDetail As String
    errDetail = "GET Error: " & err.Number & " - " & err.Description
    Debug.Print "[httpGet] " & errDetail
    
    If Not SilentMode Then
        Select Case err.Number
            Case -2147012889
                MsgBoxUni "Loi ket noi Server hoac sai dia chi Web", vbExclamation, "Thông báo - Http get error"
            Case -2147012894
                MsgBoxUni "Qu" & ChrW(225) & " th" & ChrW(7901) & "i gian k" & ChrW(7871) & "t n" & ChrW(7889) & "i v" & ChrW(7899) & "i m" & ChrW(225) & "y ch" & ChrW(7911) & " Web." & vbCrLf _
                & "Ki" & ChrW(7875) & "m tra " & ChrW(273) & ChrW(432) & ChrW(7901) & "ng truy" & ChrW(7873) & "n Internet v" & ChrW(224) & " th" & ChrW(7917) & " k" & ChrW(7871) & "t n" & ChrW(7889) & "i l" & ChrW(7841) & "i.", vbExclamation, "Thông báo - Http get error"
            Case Else
                MsgBox "ErrNum: " & err.Number & vbCrLf & "Err description: " & err.Description, vbExclamation, "Thông báo - Http Get error"
        End Select
    End If
    
    If Len(sMST) > 0 Then
        Call WriteAuditLog("ERROR", "HTTP_GET_EXCEPTION", errDetail & " (URL: " & url & ")")
    End If
    
    httpGet = False
    Resume EH_Exit
End Function

' ===========================================================================
' Legacy HTTP GET Binary download helper
' ===========================================================================
Function httpGet_ex(ByVal url As String, bearer As String, Optional ByVal SilentMode As Boolean = False) As Boolean
    On Error GoTo EH
    
    Dim client As Object
    Set client = GetxmlHttp()
    
    With client
        .Open "GET", url, False
        .setRequestHeader "User-Agent", "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36"
        .setRequestHeader "Accept", "application/zip,application/octet-stream,*/*"
        .setRequestHeader "Authorization", "Bearer " & bearer
        .send

        Dim stream As Object
        Set stream = CreateObject("ADODB.Stream")
        stream.Type = 1 ' Binary
        stream.Open
        stream.Write .responseBody
        
        Dim savePath As String
        savePath = Environ$("USERPROFILE") & "\Documents\file_" & Format(Now, "yymmdd_hhnnss") & ".zip"
        stream.SaveToFile savePath, 2 ' Overwrite
        stream.Close
        
        getStatus = .Status
        If getStatus <> "300" And getStatus <> "200" Then
            Debug.Print "[httpGet_ex] HTTP " & getStatus
            httpGet_ex = False
        Else
            httpGet_ex = True
        End If
    End With
    
EH_Exit:
    Exit Function
    
EH:
    Debug.Print "[httpGet_ex] Error: " & err.Number & " - " & err.Description
    httpGet_ex = False
    Resume EH_Exit
End Function

Public Function validateInternetConnection() As Boolean
    Dim internetStatus As Boolean
    internetStatus = InternetGetConnectedState(0&, 0&)
    If internetStatus = False Then
        MsgBoxUni "Ki" & ChrW(7875) & "m tra l" & ChrW(7841) & "i k" & ChrW(7871) & "t n" & ChrW(7889) & "i internet", vbExclamation, "Thông báo"
        validateInternetConnection = False
        Exit Function
    End If
    validateInternetConnection = True
End Function

Public Function GetInternetConnectedState() As Boolean
    GetInternetConnectedState = InternetGetConnectedState(0&, 0&)
End Function
